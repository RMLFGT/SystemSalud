/* =====================================================
   1. TABLA ROL
   ===================================================== */

CREATE TABLE rol (
    id_rol        NUMBER GENERATED BY DEFAULT AS IDENTITY,
    nombre        VARCHAR2(50)  NOT NULL,
    descripcion   VARCHAR2(200),
    estado        VARCHAR2(10)  DEFAULT 'ACTIVO' NOT NULL,

    CONSTRAINT pk_rol
        PRIMARY KEY (id_rol),

    CONSTRAINT uk_rol_nombre
        UNIQUE (nombre),

    CONSTRAINT ck_rol_estado
        CHECK (estado IN ('ACTIVO', 'INACTIVO'))
);


/* =====================================================
   2. TABLA USUARIO
   ===================================================== */

CREATE TABLE usuario (
    id_usuario       NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_rol           NUMBER         NOT NULL,
    nombre           VARCHAR2(60)   NOT NULL,
    apellido         VARCHAR2(60)   NOT NULL,
    nombre_usuario   VARCHAR2(50)   NOT NULL,
    contrasena_hash  VARCHAR2(255)  NOT NULL,
    correo           VARCHAR2(120)  NOT NULL,
    estado           VARCHAR2(10)   DEFAULT 'ACTIVO' NOT NULL,
    fecha_creacion   TIMESTAMP      DEFAULT SYSTIMESTAMP NOT NULL,
    ultimo_acceso    TIMESTAMP,

    CONSTRAINT pk_usuario
        PRIMARY KEY (id_usuario),

    CONSTRAINT uk_usuario_nombre
        UNIQUE (nombre_usuario),

    CONSTRAINT uk_usuario_correo
        UNIQUE (correo),

    CONSTRAINT ck_usuario_estado
        CHECK (estado IN ('ACTIVO', 'INACTIVO', 'BLOQUEADO')),

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (id_rol)
        REFERENCES rol (id_rol)
);


/* =====================================================
   3. TABLA PACIENTE
   ===================================================== */

CREATE TABLE paciente (
    id_paciente          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    dpi                  VARCHAR2(13)   NOT NULL,
    nombre               VARCHAR2(60)   NOT NULL,
    apellido             VARCHAR2(60)   NOT NULL,
    fecha_nacimiento     DATE           NOT NULL,
    sexo                 VARCHAR2(10)   NOT NULL,
    tipo_sangre          VARCHAR2(3),
    telefono             VARCHAR2(20),
    correo               VARCHAR2(120),
    direccion            VARCHAR2(250),
    contacto_emergencia  VARCHAR2(120),
    telefono_emergencia  VARCHAR2(20),
    estado               VARCHAR2(10)   DEFAULT 'ACTIVO' NOT NULL,
    fecha_registro       TIMESTAMP      DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_paciente
        PRIMARY KEY (id_paciente),

    CONSTRAINT uk_paciente_dpi
        UNIQUE (dpi),

    CONSTRAINT ck_paciente_dpi
        CHECK (REGEXP_LIKE(dpi, '^[0-9]{13}$')),

    CONSTRAINT ck_paciente_sexo
        CHECK (sexo IN ('MASCULINO', 'FEMENINO')),

    CONSTRAINT ck_paciente_sangre
        CHECK (
            tipo_sangre IS NULL
            OR tipo_sangre IN (
                'O+', 'O-',
                'A+', 'A-',
                'B+', 'B-',
                'AB+', 'AB-'
            )
        ),

    CONSTRAINT ck_paciente_estado
        CHECK (estado IN ('ACTIVO', 'INACTIVO'))
);

/* =====================================================
   6. TABLA HORARIO_MEDICO
   ===================================================== */

CREATE TABLE horario_medico (
    id_horario     NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_medico      NUMBER        NOT NULL,
    dia_semana     VARCHAR2(10)  NOT NULL,
    hora_inicio    VARCHAR2(5)   NOT NULL,
    hora_fin       VARCHAR2(5)   NOT NULL,
    estado         VARCHAR2(10)  DEFAULT 'ACTIVO' NOT NULL,

    CONSTRAINT pk_horario_medico
        PRIMARY KEY (id_horario),

    CONSTRAINT fk_horario_medico_medico
        FOREIGN KEY (id_medico)
        REFERENCES medico (id_medico),

    CONSTRAINT uk_horario_medico
        UNIQUE (
            id_medico,
            dia_semana,
            hora_inicio,
            hora_fin
        ),

    CONSTRAINT ck_horario_dia
        CHECK (
            dia_semana IN (
                'LUNES',
                'MARTES',
                'MIERCOLES',
                'JUEVES',
                'VIERNES',
                'SABADO',
                'DOMINGO'
            )
        ),

    CONSTRAINT ck_horario_inicio
        CHECK (
            REGEXP_LIKE(
                hora_inicio,
                '^([01][0-9]|2[0-3]):[0-5][0-9]$'
            )
        ),

    CONSTRAINT ck_horario_fin
        CHECK (
            REGEXP_LIKE(
                hora_fin,
                '^([01][0-9]|2[0-3]):[0-5][0-9]$'
            )
        ),

    CONSTRAINT ck_horario_rango
        CHECK (hora_inicio < hora_fin),

    CONSTRAINT ck_horario_estado
        CHECK (estado IN ('ACTIVO', 'INACTIVO'))
);


/* =====================================================
   7. TABLA CITA
   ===================================================== */

CREATE TABLE cita (
    id_cita          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_paciente      NUMBER         NOT NULL,
    id_medico        NUMBER         NOT NULL,
    fecha_hora       TIMESTAMP      NOT NULL,
    motivo           VARCHAR2(250)  NOT NULL,
    notas             VARCHAR2(500),
    estado            VARCHAR2(12)  DEFAULT 'PENDIENTE' NOT NULL,
    fecha_registro    TIMESTAMP     DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_cita
        PRIMARY KEY (id_cita),

    CONSTRAINT fk_cita_paciente
        FOREIGN KEY (id_paciente)
        REFERENCES paciente (id_paciente),

    CONSTRAINT fk_cita_medico
        FOREIGN KEY (id_medico)
        REFERENCES medico (id_medico),

    CONSTRAINT uk_cita_medico_fecha
        UNIQUE (id_medico, fecha_hora),

    CONSTRAINT uk_cita_paciente_fecha
        UNIQUE (id_paciente, fecha_hora),

    CONSTRAINT ck_cita_estado
        CHECK (
            estado IN (
                'PENDIENTE',
                'CONFIRMADA',
                'CANCELADA',
                'ATENDIDA'
            )
        )
);

/* =====================================================
   10. TABLA CONSULTA_MEDICA
   ===================================================== */

CREATE TABLE consulta_medica (
    id_consulta            NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_expediente          NUMBER         NOT NULL,
    id_medico              NUMBER         NOT NULL,
    id_cita                NUMBER,
    id_tipo_consulta       NUMBER         NOT NULL,
    fecha_consulta         TIMESTAMP      DEFAULT SYSTIMESTAMP NOT NULL,
    diagnostico            VARCHAR2(500)  NOT NULL,
    sintomas               VARCHAR2(2000),
    tratamiento            VARCHAR2(2000),
    observaciones          VARCHAR2(2000),
    presion_sistolica      NUMBER(3),
    presion_diastolica     NUMBER(3),
    frecuencia_cardiaca    NUMBER(3),
    temperatura            NUMBER(4,1),
    peso                   NUMBER(5,2),
    estado                 VARCHAR2(10)   DEFAULT 'FINALIZADA' NOT NULL,

    CONSTRAINT pk_consulta_medica
        PRIMARY KEY (id_consulta),

    CONSTRAINT uk_consulta_cita
        UNIQUE (id_cita),

    CONSTRAINT fk_consulta_expediente
        FOREIGN KEY (id_expediente)
        REFERENCES expediente (id_expediente),

    CONSTRAINT fk_consulta_medico
        FOREIGN KEY (id_medico)
        REFERENCES medico (id_medico),

    CONSTRAINT fk_consulta_cita
        FOREIGN KEY (id_cita)
        REFERENCES cita (id_cita),

    CONSTRAINT fk_consulta_tipo
        FOREIGN KEY (id_tipo_consulta)
        REFERENCES tipo_consulta (id_tipo_consulta),

    CONSTRAINT ck_consulta_sistolica
        CHECK (
            presion_sistolica IS NULL
            OR presion_sistolica BETWEEN 50 AND 250
        ),

    CONSTRAINT ck_consulta_diastolica
        CHECK (
            presion_diastolica IS NULL
            OR presion_diastolica BETWEEN 30 AND 150
        ),

    CONSTRAINT ck_consulta_frecuencia
        CHECK (
            frecuencia_cardiaca IS NULL
            OR frecuencia_cardiaca BETWEEN 30 AND 250
        ),

    CONSTRAINT ck_consulta_temperatura
        CHECK (
            temperatura IS NULL
            OR temperatura BETWEEN 30 AND 45
        ),

    CONSTRAINT ck_consulta_peso
        CHECK (
            peso IS NULL
            OR peso BETWEEN 1 AND 500
        ),

    CONSTRAINT ck_consulta_estado
        CHECK (
            estado IN (
                'ABIERTA',
                'FINALIZADA',
                'ANULADA'
            )
        )
);

/* =====================================================
   12. TABLA ESTUDIO_LABORATORIO
   ===================================================== */

CREATE TABLE estudio_laboratorio (
    id_estudio         NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_paciente        NUMBER          NOT NULL,
    id_medico          NUMBER          NOT NULL,
    id_consulta        NUMBER,
    id_tipo_estudio    NUMBER          NOT NULL,
    fecha_solicitud    TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    fecha_resultado    TIMESTAMP,
    resultado          VARCHAR2(4000),
    observaciones      VARCHAR2(1000),
    estado             VARCHAR2(12)    DEFAULT 'PENDIENTE' NOT NULL,

    CONSTRAINT pk_estudio_laboratorio
        PRIMARY KEY (id_estudio),

    CONSTRAINT fk_estudio_paciente
        FOREIGN KEY (id_paciente)
        REFERENCES paciente (id_paciente),

    CONSTRAINT fk_estudio_medico
        FOREIGN KEY (id_medico)
        REFERENCES medico (id_medico),

    CONSTRAINT fk_estudio_consulta
        FOREIGN KEY (id_consulta)
        REFERENCES consulta_medica (id_consulta),

    CONSTRAINT fk_estudio_tipo
        FOREIGN KEY (id_tipo_estudio)
        REFERENCES tipo_estudio (id_tipo_estudio),

    CONSTRAINT ck_estudio_estado
        CHECK (
            estado IN (
                'PENDIENTE',
                'EN_PROCESO',
                'COMPLETADO',
                'CANCELADO'
            )
        ),

    CONSTRAINT ck_estudio_fechas
        CHECK (
            fecha_resultado IS NULL
            OR fecha_resultado >= fecha_solicitud
        ),

    CONSTRAINT ck_estudio_completado
        CHECK (
            estado <> 'COMPLETADO'
            OR (
                resultado IS NOT NULL
                AND fecha_resultado IS NOT NULL
            )
        )
);

/* =====================================================
   14. TABLA MEDICAMENTO
   ===================================================== */

CREATE TABLE medicamento (
    id_medicamento          NUMBER GENERATED BY DEFAULT AS IDENTITY,
    id_categoria            NUMBER          NOT NULL,
    nombre                  VARCHAR2(120)   NOT NULL,
    presentacion            VARCHAR2(120)   NOT NULL,
    laboratorio_fabricante  VARCHAR2(120),
    stock                   NUMBER(10)      DEFAULT 0 NOT NULL,
    stock_minimo            NUMBER(10)      DEFAULT 0 NOT NULL,
    precio                  NUMBER(10,2)    NOT NULL,
    fecha_vencimiento       DATE            NOT NULL,
    descripcion             VARCHAR2(1000),
    estado                  VARCHAR2(10)     DEFAULT 'ACTIVO' NOT NULL,
    fecha_registro          TIMESTAMP        DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_medicamento
        PRIMARY KEY (id_medicamento),

    CONSTRAINT fk_medicamento_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria_medicamento (id_categoria),

    CONSTRAINT uk_medicamento_producto
        UNIQUE (
            nombre,
            presentacion,
            laboratorio_fabricante
        ),

    CONSTRAINT ck_medicamento_stock
        CHECK (stock >= 0),

    CONSTRAINT ck_medicamento_stock_minimo
        CHECK (stock_minimo >= 0),

    CONSTRAINT ck_medicamento_precio
        CHECK (precio >= 0),

    CONSTRAINT ck_medicamento_estado
        CHECK (estado IN ('ACTIVO', 'INACTIVO'))
);
