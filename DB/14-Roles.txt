/* ============================================================
   1. CREAR LA TABLA DE MÓDULOS
   Ejecutar una sola vez.
   ============================================================ */

CREATE TABLE MODULO (
    ID_MODULO NUMBER NOT NULL,
    NOMBRE VARCHAR2(50) NOT NULL,
    RUTA VARCHAR2(150) NOT NULL,
    ESTADO VARCHAR2(10) DEFAULT 'ACTIVO' NOT NULL,

    CONSTRAINT PK_MODULO
        PRIMARY KEY (ID_MODULO),

    CONSTRAINT UK_MODULO_NOMBRE
        UNIQUE (NOMBRE),

    CONSTRAINT CK_MODULO_ESTADO
        CHECK (ESTADO IN ('ACTIVO', 'INACTIVO'))
);


/* ============================================================
   2. CREAR LA TABLA DE PERMISOS POR ROL
   Ejecutar una sola vez.
   ============================================================ */

CREATE TABLE ROL_MODULO (
    ID_ROL NUMBER NOT NULL,
    ID_MODULO NUMBER NOT NULL,
    PUEDE_VER NUMBER(1) DEFAULT 0 NOT NULL,
    PUEDE_CREAR NUMBER(1) DEFAULT 0 NOT NULL,
    PUEDE_EDITAR NUMBER(1) DEFAULT 0 NOT NULL,
    PUEDE_ELIMINAR NUMBER(1) DEFAULT 0 NOT NULL,

    CONSTRAINT PK_ROL_MODULO
        PRIMARY KEY (ID_ROL, ID_MODULO),

    CONSTRAINT FK_ROL_MODULO_ROL
        FOREIGN KEY (ID_ROL)
        REFERENCES ROL (ID_ROL),

    CONSTRAINT FK_ROL_MODULO_MODULO
        FOREIGN KEY (ID_MODULO)
        REFERENCES MODULO (ID_MODULO),

    CONSTRAINT CK_RM_VER
        CHECK (PUEDE_VER IN (0, 1)),

    CONSTRAINT CK_RM_CREAR
        CHECK (PUEDE_CREAR IN (0, 1)),

    CONSTRAINT CK_RM_EDITAR
        CHECK (PUEDE_EDITAR IN (0, 1)),

    CONSTRAINT CK_RM_ELIMINAR
        CHECK (PUEDE_ELIMINAR IN (0, 1))
);

/* Contar módulos y permisos. Debe devolver 9 y 18. */

SELECT COUNT(*) AS TOTAL_MODULOS
FROM MODULO;

SELECT COUNT(*) AS TOTAL_PERMISOS
FROM ROL_MODULO;


/* Consultar la matriz completa de permisos. */

SELECT
    r.ID_ROL,
    r.NOMBRE AS ROL,
    m.ID_MODULO,
    m.NOMBRE AS MODULO,
    rm.PUEDE_VER,
    rm.PUEDE_CREAR,
    rm.PUEDE_EDITAR,
    rm.PUEDE_ELIMINAR
FROM ROL_MODULO rm
INNER JOIN ROL r
    ON r.ID_ROL = rm.ID_ROL
INNER JOIN MODULO m
    ON m.ID_MODULO = rm.ID_MODULO
ORDER BY r.ID_ROL, m.ID_MODULO;


/* Consultar los módulos visibles para un rol concreto.
   Cambia :ID_ROL por 2, 3, 4 o 5 al probar manualmente. */

SELECT
    m.NOMBRE AS MODULO,
    m.RUTA,
    rm.PUEDE_CREAR,
    rm.PUEDE_EDITAR,
    rm.PUEDE_ELIMINAR
FROM ROL_MODULO rm
INNER JOIN MODULO m
    ON m.ID_MODULO = rm.ID_MODULO
WHERE rm.ID_ROL = :ID_ROL
  AND rm.PUEDE_VER = 1
  AND m.ESTADO = 'ACTIVO'
ORDER BY m.ID_MODULO;
