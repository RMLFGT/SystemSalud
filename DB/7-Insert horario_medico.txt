INSERT INTO horario_medico (
    id_medico, id_clinica, dia_semana, hora_inicio, hora_fin
)
SELECT
    m.id_medico,
    c.id_clinica,
    d.dia_semana,
    '08:00',
    '14:00'
FROM medico m
CROSS JOIN clinica c
CROSS JOIN (
    SELECT 'LUNES' dia_semana FROM dual
    UNION ALL SELECT 'MARTES' FROM dual
    UNION ALL SELECT 'MIERCOLES' FROM dual
    UNION ALL SELECT 'JUEVES' FROM dual
    UNION ALL SELECT 'VIERNES' FROM dual
) d
WHERE m.numero_colegiado = '45821'
AND c.codigo_clinica = '1';
